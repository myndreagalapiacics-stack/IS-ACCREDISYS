# IS ACCREDISYS — Project Foundation


Foundation codebase derived from the group's **SRS**, **SPMP**, and project notes
(Information Systems Accreditation Compliance and Evidence Management System, UST CICS, 3ISC Group 4).

## Run
```bash
pip install -r requirements.txt        # Flask and QR code generation
python seed.py                         # creates DB + SAMPLE data, prints demo TOTP secrets
flask --app app run                    # http://127.0.0.1:5000
python -m unittest discover -s tests -t .
```
Set `ACCREDISYS_SECRET` in any real deployment. Use HTTPS (SRS 3.4).

Version history records are retained, but file contents are kept only for the two
most recent approved versions, the most recent pending upload, and the most
recent rejected upload. Older entries remain visible in history as unavailable.
Run `flask --app app prune-version-files` once to apply this policy to files
already present in the database.

## Stack decision (SPMP says "not yet confirmed")
Python/Flask + SQLite, chosen for ₱0 cost (SPMP Table C-1) and lightweight dependencies.
Schema is plain SQL, so moving to PostgreSQL/MySQL (SRS 3.3) is a small change. **Confirm with your adviser.**

## What is built vs. SRS
| SRS item | Where | Status |
|---|---|---|
| FR-01, BR-6 password + authenticator (TOTP, RFC 6238) | `auth.py`, `security.py` | Done, tested vs RFC vector; account creation shows a local QR code and one-time setup key |
| FR-02, BR-1 RBAC | `security.PERMISSIONS` | Done (Chair, Faculty, Evaluator) |
| FR-03/04 upload + metadata (FR-03.2) | `repository.py` | Done; exact metadata matches are uploaded as new versions, and owners/Chair can submit replacements from the repository |
| FR-05 search & filters | `repository.search` | Done |
| FR-10, BR-3/4/5 versioning, Pending Review, remarks, no overwrite/delete | `repository.py`, `approvals.py`, DB triggers | Done |
| FR-07 approval + signatories | `approvals.py` | Done |
| FR-06 requirement tracking, auto-status on approval | `requirements.py` | Done |
| FR-08, BR-7 faculty profile/history, privacy scoping | `faculty.py` | Done (row-level); profile edits are in Profile and Faculty/Teaching Information is read-only |
| FR-09 dashboard + CSV report | `dashboard.py` | Done (CSV; PDF later) |
| FR-11, BR-8 backup/restore | `admin.py` | Done (call `make_backup()` from cron for scheduled) |
| FR-12, BR-2 account lifecycle | `admin.py` | Done (soft deactivate) |

## PDF annotations and file previews

PDF highlights and comment threads are stored in the database and do not modify
uploaded originals. Open a retained document version from its version history to
view the PDF and add or manage comments. The owner and Department Chair can see
comments; other permitted viewers can read the file without seeing them.

Word, PowerPoint, Excel, text, and image uploads can be converted to PDF previews
with LibreOffice. Install LibreOffice on the server for conversion; PDF viewing
and annotation do not require it. Previews are stored with uploads and included
in backups. Install the PDF export dependency with `pip install -r requirements.txt`.

-- IS ACCREDISYS database schema (SRS FR-01..FR-12, BR-1..BR-10)
PRAGMA foreign_keys = ON;

-- FR-01, FR-02, FR-12, BR-1, BR-2, BR-6, BR-10
CREATE TABLE IF NOT EXISTS users (
  id              INTEGER PRIMARY KEY,
  email           TEXT NOT NULL UNIQUE,
  full_name       TEXT NOT NULL,
  password_hash   TEXT NOT NULL,
  role            TEXT NOT NULL CHECK (role IN ('dept_chair','faculty','external_evaluator')),
  totp_secret     TEXT NOT NULL,                 -- BR-6: authenticator (TOTP) is mandatory
  is_active       INTEGER NOT NULL DEFAULT 1,    -- BR-2: deactivation instead of deletion
  expires_at      TEXT,                          -- BR-10: temporary evaluator accounts
  failed_logins   INTEGER NOT NULL DEFAULT 0,
  locked_until    TEXT,
  created_at      TEXT NOT NULL DEFAULT (datetime('now')),
  deactivated_at  TEXT
);

-- FR-03 / FR-05: document metadata used for search and filtering
CREATE TABLE IF NOT EXISTS categories (
  id        INTEGER PRIMARY KEY,
  name      TEXT NOT NULL,
  parent_id INTEGER REFERENCES categories(id),   -- main categories -> subcategories (interview 9/11)
  UNIQUE(name, parent_id)
);

CREATE TABLE IF NOT EXISTS courses (
  id    INTEGER PRIMARY KEY,
  code  TEXT NOT NULL UNIQUE,
  title TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS documents (
  id               INTEGER PRIMARY KEY,
  title            TEXT NOT NULL,
  description      TEXT NOT NULL DEFAULT '',
  category_id      INTEGER NOT NULL REFERENCES categories(id),
  course_id        INTEGER REFERENCES courses(id),
  school_year      TEXT NOT NULL,                -- e.g. 2026-2027
  semester         TEXT NOT NULL CHECK (semester IN ('1st','2nd','Summer')),
  document_type    TEXT NOT NULL,                -- syllabus, exam, credential, ...
  owner_id         INTEGER NOT NULL REFERENCES users(id),   -- BR-9 data owner
  current_version_id INTEGER,                    -- latest *approved* version (BR-3)
  created_at       TEXT NOT NULL DEFAULT (datetime('now'))
);

-- Version records remain append-only; older file contents may be pruned while audit history is kept.
CREATE TABLE IF NOT EXISTS document_versions (
  id             INTEGER PRIMARY KEY,
  document_id    INTEGER NOT NULL REFERENCES documents(id),
  version_no     INTEGER NOT NULL,
  stored_path    TEXT NOT NULL,
  original_name  TEXT NOT NULL,
  sha256         TEXT NOT NULL,
  size_bytes     INTEGER NOT NULL,
  status         TEXT NOT NULL DEFAULT 'Pending Review'
                 CHECK (status IN ('Pending Review','Current Approved Version','Requires Revision','Superseded')),
  remarks        TEXT,
  uploaded_by    INTEGER NOT NULL REFERENCES users(id),
  uploaded_at    TEXT NOT NULL DEFAULT (datetime('now')),
  reviewed_by    INTEGER REFERENCES users(id),
  reviewed_at    TEXT,
  UNIQUE(document_id, version_no)
);

-- Keep version records and file metadata immutable; retention removes file contents from disk only.
CREATE TRIGGER IF NOT EXISTS trg_versions_no_delete
BEFORE DELETE ON document_versions
BEGIN SELECT RAISE(ABORT, 'BR-5: document versions cannot be deleted'); END;

CREATE TRIGGER IF NOT EXISTS trg_versions_immutable_file
BEFORE UPDATE OF stored_path, sha256, version_no, document_id, original_name, size_bytes ON document_versions
BEGIN SELECT RAISE(ABORT, 'BR-5: stored version content is immutable'); END;

-- FR-07: signatories per document
CREATE TABLE IF NOT EXISTS signatories (
  id          INTEGER PRIMARY KEY,
  document_id INTEGER NOT NULL REFERENCES documents(id),
  version_id  INTEGER NOT NULL REFERENCES document_versions(id),
  signer_id   INTEGER NOT NULL REFERENCES users(id),
  signed_at   TEXT,
  UNIQUE(version_id, signer_id)
);

-- FR-06: accreditation requirement tracking
CREATE TABLE IF NOT EXISTS accreditation_bodies (
  id   INTEGER PRIMARY KEY,
  name TEXT NOT NULL UNIQUE                      -- e.g. PAASCU, CHED (names from interview 10/9)
);

CREATE TABLE IF NOT EXISTS requirements (
  id          INTEGER PRIMARY KEY,
  body_id     INTEGER NOT NULL REFERENCES accreditation_bodies(id),
  code        TEXT NOT NULL,
  description TEXT NOT NULL,
  status      TEXT NOT NULL DEFAULT 'Missing Evidence'
              CHECK (status IN ('Completed','In Progress','Missing Evidence')),
  UNIQUE(body_id, code)
);

CREATE TABLE IF NOT EXISTS requirement_evidence (
  requirement_id INTEGER NOT NULL REFERENCES requirements(id),
  document_id    INTEGER NOT NULL REFERENCES documents(id),
  linked_by      INTEGER NOT NULL REFERENCES users(id),
  linked_at      TEXT NOT NULL DEFAULT (datetime('now')),
  PRIMARY KEY (requirement_id, document_id)
);

-- FR-08, BR-7: faculty profile data (privacy-sensitive)
CREATE TABLE IF NOT EXISTS faculty_profiles (
  id              INTEGER PRIMARY KEY,
  user_id         INTEGER UNIQUE REFERENCES users(id),
  employment_type TEXT NOT NULL DEFAULT 'Regular' CHECK (employment_type IN ('Regular','Part-time')),
  highest_degree  TEXT,                          -- doctoral / masteral / pursuing (interview 10/8)
  specialization  TEXT,
  photo_path      TEXT,
  address         TEXT,
  contact_number  TEXT,
  facebook_name   TEXT,
  ust_email       TEXT,
  faculty_status  TEXT NOT NULL DEFAULT 'Full-time' CHECK (faculty_status IN ('Full-time','Part-time')),
  date_started    TEXT
);

CREATE TABLE IF NOT EXISTS faculty_courses (
  faculty_id INTEGER NOT NULL REFERENCES faculty_profiles(id) ON DELETE CASCADE,
  course_id  INTEGER NOT NULL REFERENCES courses(id) ON DELETE CASCADE,
  PRIMARY KEY (faculty_id, course_id)
);

CREATE TABLE IF NOT EXISTS faculty_records (
  id          INTEGER PRIMARY KEY,
  faculty_id  INTEGER NOT NULL REFERENCES faculty_profiles(id),
  kind        TEXT NOT NULL CHECK (kind IN ('education','credential','award','research','activity')),
  title       TEXT NOT NULL,
  detail      TEXT,
  year        INTEGER,
  scope       TEXT NOT NULL DEFAULT 'international' CHECK (scope IN ('local','international')),
  created_at  TEXT NOT NULL DEFAULT (datetime('now'))
);

-- Evaluator access requests (BR-10)
CREATE TABLE IF NOT EXISTS access_requests (
  id            INTEGER PRIMARY KEY,
  evaluator_id  INTEGER NOT NULL REFERENCES users(id),
  document_id   INTEGER NOT NULL REFERENCES documents(id),
  status        TEXT NOT NULL DEFAULT 'Pending' CHECK (status IN ('Pending','Approved','Denied')),
  decided_by    INTEGER REFERENCES users(id),
  decided_at    TEXT,
  UNIQUE(evaluator_id, document_id)
);

-- Update/correction requests from the Chair to faculty (use case: Request updates or corrections)
CREATE TABLE IF NOT EXISTS revision_requests (
  id           INTEGER PRIMARY KEY,
  document_id  INTEGER NOT NULL REFERENCES documents(id),
  requested_by INTEGER NOT NULL REFERENCES users(id),
  assigned_to  INTEGER NOT NULL REFERENCES users(id),
  message      TEXT NOT NULL,
  resolved     INTEGER NOT NULL DEFAULT 0,
  created_at   TEXT NOT NULL DEFAULT (datetime('now'))
);

-- Department dashboard statistics (interview 9/11: enrollees, graduates per year)
CREATE TABLE IF NOT EXISTS department_stats (
  school_year TEXT NOT NULL,
  metric      TEXT NOT NULL CHECK (metric IN ('enrollees','graduates')),
  year_level  INTEGER NOT NULL DEFAULT 0,
  value       INTEGER NOT NULL,
  PRIMARY KEY (school_year, metric, year_level)
);

-- Accountability for BR-1/BR-3/BR-5 etc.
CREATE TABLE IF NOT EXISTS audit_log (
  id         INTEGER PRIMARY KEY,
  user_id    INTEGER REFERENCES users(id),
  action     TEXT NOT NULL,
  entity     TEXT,
  entity_id  INTEGER,
  detail     TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS backups (
  id         INTEGER PRIMARY KEY,
  path       TEXT NOT NULL,
  kind       TEXT NOT NULL CHECK (kind IN ('scheduled','manual')),
  created_by INTEGER REFERENCES users(id),
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE INDEX IF NOT EXISTS idx_docs_filter ON documents(category_id, course_id, school_year, semester, document_type);
CREATE INDEX IF NOT EXISTS idx_docs_title ON documents(title);
CREATE INDEX IF NOT EXISTS idx_versions_doc ON document_versions(document_id, version_no);

-- Annotations are stored separately so the original version file remains unchanged.
CREATE TABLE IF NOT EXISTS annotations (
  id          INTEGER PRIMARY KEY,
  version_id  INTEGER NOT NULL REFERENCES document_versions(id),
  parent_id   INTEGER REFERENCES annotations(id) ON DELETE CASCADE,
  page        INTEGER,
  rects       TEXT,
  quote       TEXT NOT NULL DEFAULT '',
  comment     TEXT NOT NULL DEFAULT '',
  author_id   INTEGER NOT NULL REFERENCES users(id),
  resolved    INTEGER NOT NULL DEFAULT 0,
  created_at  TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at  TEXT NOT NULL DEFAULT (datetime('now')),
  CHECK (parent_id IS NOT NULL OR (page IS NOT NULL AND rects IS NOT NULL))
);
CREATE INDEX IF NOT EXISTS idx_annotations_version ON annotations(version_id, page);

CREATE TABLE IF NOT EXISTS version_previews (
  version_id INTEGER PRIMARY KEY REFERENCES document_versions(id),
  status     TEXT NOT NULL CHECK (status IN ('pending','ready','failed','unavailable')),
  path       TEXT,
  sha256     TEXT,
  error      TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

import mimetypes, os
from datetime import datetime, timezone
from pathlib import Path
from flask import Flask, g, jsonify, request, session
from . import db, security


def _expired(user) -> bool:
    if not user["expires_at"]:
        return False
    return datetime.fromisoformat(user["expires_at"]) < datetime.now(timezone.utc).replace(tzinfo=None)


def create_app(config: dict | None = None) -> Flask:
    app = Flask(__name__, instance_relative_config=True)
    inst = Path(app.instance_path)
    app.config.update(
        SECRET_KEY=os.environ.get("ACCREDISYS_SECRET", "dev-only-change-me"),
        DISABLE_TOTP=False,
        DATABASE=str(inst / "accredisys.sqlite"),
        UPLOAD_DIR=str(inst / "uploads"),
        BACKUP_DIR=str(inst / "backups"),
        MAX_CONTENT_LENGTH=25 * 1024 * 1024,
        ALLOWED_EXTENSIONS={"pdf", "docx", "xlsx", "pptx", "png", "jpg", "jpeg", "txt"},
        SESSION_COOKIE_HTTPONLY=True, SESSION_COOKIE_SAMESITE="Lax",
        MAX_FAILED_LOGINS=5, LOCKOUT_MINUTES=15,
        CONVERT_ENABLED=True, CONVERT_SYNC=False, CONVERT_TIMEOUT=120,
    )
    if config:
        app.config.update(config)
    for d in (Path(app.config["DATABASE"]).parent, app.config["UPLOAD_DIR"], app.config["BACKUP_DIR"]):
        Path(d).mkdir(parents=True, exist_ok=True)
    mimetypes.add_type("text/javascript", ".mjs")
    db.init_app(app)
    with app.app_context():
        db.init_db()

    @app.before_request
    def load_user():
        g.user = None
        uid = session.get("uid")
        if uid and session.get("mfa_ok"):
            row = db.get_db().execute("SELECT * FROM users WHERE id=?", (uid,)).fetchone()
            # BR-2: deactivated/expired accounts lose access immediately
            if row and row["is_active"] and not _expired(row):
                g.user = row

    @app.before_request
    def csrf_guard():
        if request.endpoint not in (None, "static", "auth.login", "auth.verify_mfa", "health"):
            security.check_csrf()

    @app.after_request
    def headers(resp):
        resp.headers["X-Content-Type-Options"] = "nosniff"
        resp.headers["X-Frame-Options"] = "DENY"
        resp.headers["Cache-Control"] = "no-store"
        return resp

    from .blueprints import auth, repository, approvals, requirements, faculty, admin, dashboard, evaluator, annotations
    for m in (auth, repository, approvals, requirements, faculty, admin, dashboard, evaluator, annotations):
        app.register_blueprint(m.bp)

    @app.get("/health")
    def health():
        return jsonify(status="ok")

    @app.errorhandler(403)
    def forbidden(e):  # NFR-11: understandable message, no technical detail
        return jsonify(error="You do not have permission to do that."), 403

    @app.errorhandler(401)
    def unauth(e):
        return jsonify(error="Please sign in."), 401

    @app.get("/")
    def index():
        return app.send_static_file("index.html")

    @app.errorhandler(400)
    def bad(e):
        return jsonify(error=getattr(e, "description", "Bad request")), 400

    @app.errorhandler(415)
    def unsupported(e):
        return jsonify(error="Preview is available for PDF files only."), 415

    @app.errorhandler(404)
    def nf(e):
        return jsonify(error="Record not found."), 404

    return app

"""Phase 2: PDF previews for Word / PowerPoint / Excel / text / image files, made with LibreOffice.

Why: the viewer and comments work on PDF. A converted preview is stored ONCE per version and never
regenerated while comments exist, because comment positions are anchored to its layout. The original
file is untouched (BR-5). Previews live under UPLOAD_DIR/previews so backups include them."""
import hashlib, os, queue, shutil, subprocess, tempfile, threading
from pathlib import Path
from flask import current_app
from .db import get_db

CONVERTIBLE = {"docx", "pptx", "xlsx", "txt", "png", "jpg", "jpeg"}
FILTERS = {  # Excel: one page per sheet, so wide sheets aren't sliced across pages
    "xlsx": 'pdf:calc_pdf_Export:{"SinglePageSheets":{"type":"boolean","value":"true"}}'}
_jobs: queue.Queue = queue.Queue()
_worker = None
_wlock = threading.Lock()
_version_cache: dict = {}


def ext_of(name: str) -> str:
    return name.rsplit(".", 1)[-1].lower() if "." in name else ""


def is_pdf(path) -> bool:
    try:
        with open(path, "rb") as f:
            return b"%PDF-" in f.read(1024)
    except OSError:
        return False


def soffice_path():
    env = os.environ.get("ACCREDISYS_SOFFICE")
    if env and Path(env).exists():
        return env
    for n in ("soffice", "libreoffice"):
        p = shutil.which(n)
        if p:
            return p
    for p in (r"C:\Program Files\LibreOffice\program\soffice.exe", r"C:\Program Files (x86)\LibreOffice\program\soffice.exe",
              "/Applications/LibreOffice.app/Contents/MacOS/soffice"):
        if Path(p).exists():
            return p
    return None


def converter_info() -> dict:
    exe = soffice_path()
    if not exe:
        return {"available": False}
    if exe not in _version_cache:
        try:
            out = subprocess.run([exe, "--version"], capture_output=True, text=True, timeout=30).stdout.strip()
        except Exception:
            out = "LibreOffice"
        _version_cache[exe] = out or "LibreOffice"
    return {"available": True, "version": _version_cache[exe]}


def convert_file(src, ext, outdir, timeout=120) -> Path:
    """Convert one file to PDF; returns the output path. Uses a throw-away LibreOffice profile (no shared locks)."""
    exe = soffice_path()
    if not exe:
        raise RuntimeError("LibreOffice is not installed.")
    src, outdir = Path(src), Path(outdir)
    outdir.mkdir(parents=True, exist_ok=True)
    out = outdir / (src.stem + ".pdf")
    out.unlink(missing_ok=True)
    profile = Path(tempfile.mkdtemp(prefix="accredisys_lo_"))
    try:
        cmd = [exe, f"-env:UserInstallation={profile.as_uri()}", "--headless", "--norestore", "--nolockcheck",
               "--convert-to", FILTERS.get(ext, "pdf"), "--outdir", str(outdir), str(src)]
        r = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
        if not out.exists() or out.stat().st_size == 0:
            raise RuntimeError((r.stderr or r.stdout or "LibreOffice produced no output.")[:300])
        return out
    finally:
        shutil.rmtree(profile, ignore_errors=True)


# ---------- job handling ----------
def _run(app, vid, src, ext):
    with app.app_context():
        db = get_db()
        try:
            out = convert_file(src, ext, Path(app.config["UPLOAD_DIR"]) / "previews", app.config["CONVERT_TIMEOUT"])
            sha = hashlib.sha256(out.read_bytes()).hexdigest()
            db.execute("UPDATE version_previews SET status='ready', path=?, sha256=?, error=NULL, updated_at=datetime('now') WHERE version_id=?",
                       (str(out), sha, vid))
        except Exception as e:  # never let a bad file take the worker down
            db.execute("UPDATE version_previews SET status='failed', error=?, updated_at=datetime('now') WHERE version_id=?",
                       (str(e)[:300], vid))
        db.commit()


def _loop():
    while True:
        job = _jobs.get()
        try:
            _run(*job)
        except Exception:
            pass
        finally:
            _jobs.task_done()


def _enqueue(app, vid, src, ext):
    if app.config.get("CONVERT_SYNC"):
        return _run(app, vid, src, ext)
    global _worker
    with _wlock:
        if _worker is None or not _worker.is_alive():
            _worker = threading.Thread(target=_loop, daemon=True, name="preview-converter")
            _worker.start()
    _jobs.put((app, vid, src, ext))


def _set(db, vid, status):
    db.execute("INSERT INTO version_previews (version_id, status) VALUES (?, ?) ON CONFLICT(version_id) DO UPDATE SET "
               "status=excluded.status, path=NULL, sha256=NULL, error=NULL, updated_at=datetime('now')", (vid, status))
    db.commit()


def ensure_preview(vid, stored_path, original_name, force=False):
    """Make sure a preview exists or is being made. Cheap and safe to call repeatedly."""
    ext = ext_of(original_name)
    if ext not in CONVERTIBLE or not current_app.config["CONVERT_ENABLED"]:
        return
    db = get_db()
    row = db.execute("SELECT *, (updated_at < datetime('now','-10 minutes')) AS stale FROM version_previews WHERE version_id=?", (vid,)).fetchone()
    if row and not force:
        if row["status"] == "ready" and row["path"] and Path(row["path"]).exists():
            return
        if row["status"] == "pending" and not row["stale"]:
            return
        if row["status"] == "failed":
            return  # only an explicit retry re-runs a failed conversion
    if not soffice_path():
        return _set(db, vid, "unavailable")
    _set(db, vid, "pending")
    _enqueue(current_app._get_current_object(), vid, stored_path, ext)


def preview_info(vid, stored_path, original_name):
    """-> (status, pdf_path, converted). status: ready | pending | failed | unavailable | unsupported"""
    ext = ext_of(original_name)
    if ext == "pdf":
        return ("ready", stored_path, False) if is_pdf(stored_path) else ("unsupported", None, False)
    if ext not in CONVERTIBLE:
        return ("unsupported", None, False)
    if not current_app.config["CONVERT_ENABLED"]:
        return ("unavailable", None, True)
    row = get_db().execute("SELECT * FROM version_previews WHERE version_id=?", (vid,)).fetchone()
    if not row:
        return ("pending", None, True)
    if row["status"] == "ready":
        if row["path"] and Path(row["path"]).exists():
            return ("ready", row["path"], True)
        return ("pending", None, True)  # file went missing; ensure_preview will rebuild it
    return (row["status"], None, True)

"""Write annotations into a COPY of a PDF as standard highlight annotations (pypdf).
Anchors are stored as fractions of the *displayed* page (origin top-left, after /Rotate),
so they are converted back to PDF user space here. The stored original is never touched."""
from io import BytesIO


def displayed_to_pdf(box, rotate, u, v):
    """(u,v) in [0,1] of the displayed page, origin top-left -> PDF user-space (x,y)."""
    x0, y0, x1, y1 = box
    W, H = x1 - x0, y1 - y0
    r = rotate % 360
    if r == 0:
        return x0 + u * W, y1 - v * H
    if r == 90:
        return x0 + v * W, y0 + u * H
    if r == 180:
        return x1 - u * W, y0 + v * H
    return x1 - v * W, y1 - u * H  # 270


def build_annotated_pdf(data: bytes, items: list[dict]) -> tuple[bytes, int]:
    """items: [{page, rects, text, author, resolved}]. Returns (pdf_bytes, skipped_count)."""
    from pypdf import PdfReader, PdfWriter
    from pypdf.annotations import Highlight
    from pypdf.generic import ArrayObject, FloatObject, NameObject, TextStringObject

    reader = PdfReader(BytesIO(data))
    if reader.is_encrypted and not reader.decrypt(""):
        raise ValueError("This PDF is password-protected.")
    writer = PdfWriter(clone_from=reader)
    skipped = 0
    for it in items:
        n = it["page"] - 1
        if not (0 <= n < len(writer.pages)):
            skipped += 1
            continue
        page = writer.pages[n]
        cb = page.cropbox
        box = (float(cb.left), float(cb.bottom), float(cb.right), float(cb.top))
        rot = int(page.rotation or 0)
        quads, xs, ys = [], [], []
        for x, y, w, h in it["rects"]:
            for u, v in ((x, y), (x + w, y), (x, y + h), (x + w, y + h)):  # TL, TR, BL, BR
                px, py = displayed_to_pdf(box, rot, u, v)
                quads += [round(px, 3), round(py, 3)]; xs.append(px); ys.append(py)
        a = Highlight(rect=(min(xs), min(ys), max(xs), max(ys)),
                      quad_points=ArrayObject([FloatObject(q) for q in quads]),
                      highlight_color="c8c8c8" if it["resolved"] else "ffe14d")
        a[NameObject("/Contents")] = TextStringObject(it["text"])
        a[NameObject("/T")] = TextStringObject(it["author"])
        writer.add_annotation(page_number=n, annotation=a)
    out = BytesIO()
    writer.write(out)
    return out.getvalue(), skipped

"""Security primitives: TOTP (RFC 6238), CSRF, and RBAC."""
import base64, hashlib, hmac, secrets, struct, time
from io import BytesIO
from functools import wraps
from urllib.parse import quote, urlencode
from flask import abort, g, redirect, request, session, url_for
import qrcode
from qrcode.image.svg import SvgPathImage


# ---- TOTP (BR-6 / Account Security Policy) ----
def new_totp_secret() -> str:
    return base64.b32encode(secrets.token_bytes(20)).decode()


def _hotp(secret: str, counter: int, digits: int = 6) -> str:
    key = base64.b32decode(secret.upper())
    digest = hmac.new(key, struct.pack(">Q", counter), hashlib.sha1).digest()
    o = digest[-1] & 0x0F
    code = (struct.unpack(">I", digest[o:o + 4])[0] & 0x7FFFFFFF) % (10 ** digits)
    return str(code).zfill(digits)


def totp_now(secret: str, at: float | None = None) -> str:
    return _hotp(secret, int((at or time.time()) // 30))


def verify_totp(secret: str, code: str, window: int = 1) -> bool:
    code = (code or "").strip()
    if not (code.isdigit() and len(code) == 6):
        return False
    step = int(time.time() // 30)
    return any(hmac.compare_digest(_hotp(secret, step + d), code) for d in range(-window, window + 1))


def provisioning_uri(email: str, secret: str, issuer: str = "IS ACCREDISYS") -> str:
    label = f"{quote(issuer, safe='')}:{quote(email, safe='')}"
    query = urlencode({"secret": secret, "issuer": issuer, "digits": 6, "period": 30})
    return f"otpauth://totp/{label}?{query}"


def totp_qr_data_uri(uri: str) -> str:
    qr = qrcode.QRCode(error_correction=qrcode.constants.ERROR_CORRECT_M, box_size=6, border=4)
    qr.add_data(uri)
    qr.make(fit=True)
    image = qr.make_image(image_factory=SvgPathImage)
    output = BytesIO()
    image.save(output)
    encoded = base64.b64encode(output.getvalue()).decode("ascii")
    return f"data:image/svg+xml;base64,{encoded}"


# ---- CSRF ----
def csrf_token() -> str:
    if "_csrf" not in session:
        session["_csrf"] = secrets.token_urlsafe(32)
    return session["_csrf"]


def check_csrf():
    if request.method in ("POST", "PUT", "PATCH", "DELETE"):
        sent = request.headers.get("X-CSRF-Token") or request.form.get("_csrf", "")
        if not hmac.compare_digest(sent, session.get("_csrf", "")):
            abort(400, "Invalid CSRF token")

def login_required(view):
    @wraps(view)
    def wrapped(*a, **kw):
        if g.user is None:
            abort(401)
        return view(*a, **kw)
    return wrapped


def permission_required(perm: str):
    def deco(view):
        @wraps(view)
        def wrapped(*a, **kw):
            if g.user is None:
                abort(401)
            if not can(g.user, perm):
                abort(403)
            return view(*a, **kw)
        return wrapped
    return deco

from pathlib import Path


def version_file_path(version, upload_dir):
    stored_path = Path(version["stored_path"])
    if stored_path.is_file():
        return stored_path
    fallback = Path(upload_dir) / stored_path.name
    return fallback if fallback.is_file() else None


def prune_version_files(db, document_id, upload_dir):
    versions = db.execute(
        "SELECT id, version_no, status, stored_path FROM document_versions "
        "WHERE document_id=? ORDER BY version_no DESC",
        (document_id,),
    ).fetchall()
    keep = set()
    approved = 0
    pending = False
    rejected = False
    for version in versions:
        status = version["status"]
        if status in ("Current Approved Version", "Superseded") and approved < 2:
            keep.add(version["id"])
            approved += 1
        elif status == "Pending Review" and not pending:
            keep.add(version["id"])
            pending = True
        elif status == "Requires Revision" and not rejected:
            keep.add(version["id"])
            rejected = True

    removed = []
    for version in versions:
        if version["id"] in keep:
            continue
        path = version_file_path(version, upload_dir)
        if path is not None:
            path.unlink()
            removed.append(version["id"])
    return removed


def prune_all_version_files(db, upload_dir):
    document_ids = db.execute(
        "SELECT DISTINCT document_id FROM document_versions"
    ).fetchall()
    removed = []
    for row in document_ids:
        removed.extend(prune_version_files(db, row["document_id"], upload_dir))
    return removed

        out.append([round(x, 5), round(y, 5), round(w, 5), round(h, 5)])

    try:
        from ..pdf_export import build_annotated_pdf
    except ImportError:
        return jsonify(error="Exporting comments needs the 'pypdf' package (pip install -r requirements.txt)."), 501
    items = []
    for a in _list(vid):
        text = f"{a['author']}: {a['comment']}".strip() if a["comment"] else ""
        for r in a["replies"]:
            text += f"\n  ↳ {r['author']}: {r['comment']}"
        if a["resolved"]:
            text = "[Resolved] " + text
        items.append(dict(page=a["page"], rects=a["rects"], text=text.strip(), author=a["author"], resolved=a["resolved"]))
    with open(pdf_path, "rb") as f:
        data = f.read()
    try:
        out, skipped = build_annotated_pdf(data, items)
    except ValueError as e:
        abort(400, str(e))
    except Exception:
        return jsonify(error="Could not add comments to this PDF. Download the original instead."), 500
    audit("document.export_annotated", "document_version", vid, {"annotations": len(items), "skipped": skipped})
    get_db().commit()
    stem = v["original_name"].rsplit(".", 1)[0]
    return send_file(BytesIO(out), mimetype="application/pdf", as_attachment=True, download_name=f"{stem}_annotated.pdf")

"""FR-01 / FR-02 / BR-6: password + authenticator (TOTP) login."""
from datetime import datetime, timedelta, timezone
from flask import Blueprint, current_app, g, jsonify, request, session
from werkzeug.security import check_password_hash
from ..db import get_db, audit
from ..security import verify_totp, csrf_token, login_required

bp = Blueprint("auth", __name__)


def _now():
    return datetime.now(timezone.utc).replace(tzinfo=None)


@bp.post("/login")
def login():
    d = request.get_json(silent=True) or request.form
    email, pw = (d.get("email") or "").strip().lower(), d.get("password") or ""
    db = get_db()
    u = db.execute("SELECT * FROM users WHERE email=?", (email,)).fetchone()
    generic = (jsonify(error="Invalid email or password."), 401)  # NFR-11: no account enumeration
    if not u or not u["is_active"]:
        return generic
    if u["locked_until"] and datetime.fromisoformat(u["locked_until"]) > _now():
        return jsonify(error="Too many attempts. Try again later."), 429
    if not check_password_hash(u["password_hash"], pw):
        n = u["failed_logins"] + 1
        lock = None
        if n >= current_app.config["MAX_FAILED_LOGINS"]:
            lock = (_now() + timedelta(minutes=current_app.config["LOCKOUT_MINUTES"])).isoformat()
            n = 0
        db.execute("UPDATE users SET failed_logins=?, locked_until=? WHERE id=?", (n, lock, u["id"]))
        db.commit()
        return generic
    if current_app.config["DISABLE_TOTP"]:
        session.clear()
        session["uid"], session["mfa_ok"] = u["id"], True
        g.user = u
        audit("login", "user", u["id"])
        db.commit()
        return jsonify(ok=True, role=u["role"], csrf=csrf_token())
    # Step 1 passed; session is NOT authenticated until MFA succeeds (BR-6).
    session.clear()
    session["pending_uid"] = u["id"]
    return jsonify(mfa_required=True)


@bp.post("/login/verify")
def verify_mfa():
    uid = session.get("pending_uid")
    if not uid:
        return jsonify(error="Start by entering your credentials."), 401
    db = get_db()
    u = db.execute("SELECT * FROM users WHERE id=?", (uid,)).fetchone()
    code = (request.get_json(silent=True) or request.form).get("code", "")
    if not u or not verify_totp(u["totp_secret"], code):
        return jsonify(error="Invalid verification code."), 401
    session.clear()
    session["uid"], session["mfa_ok"] = u["id"], True
    db.execute("UPDATE users SET failed_logins=0, locked_until=NULL WHERE id=?", (u["id"],))
    g.user = u
    audit("login", "user", u["id"])
    db.commit()
    return jsonify(ok=True, role=u["role"], csrf=csrf_token())

def _visible_clause():
    """BR-10: evaluators only see documents with an approved access request AND an approved version."""
    if g.user["role"] == "external_evaluator":
        return (" AND d.current_version_id IS NOT NULL AND EXISTS (SELECT 1 FROM access_requests r "
                "WHERE r.document_id=d.id AND r.evaluator_id=? AND r.status='Approved')"), [g.user["id"]]
    if g.user["role"] == "faculty":
        # Faculty: own documents plus approved departmental reference files (read-only, SRS 2.2)
        return " AND (d.owner_id=? OR d.current_version_id IS NOT NULL)", [g.user["id"]]
    return "", []


@bp.get("")
@login_required
def search():
    if not (can(g.user, "repo.view") or can(g.user, "repo.view_approved_only")):
        abort(403)
    q = request.args
    sql = ("SELECT d.id, d.owner_id, d.title, d.description, d.school_year, d.semester, d.document_type, "
           "c.name AS category, co.code AS course, d.current_version_id, "
           "(SELECT id FROM document_versions WHERE document_id=d.id ORDER BY version_no DESC LIMIT 1) AS latest_version_id, "
           "(SELECT MAX(version_no) FROM document_versions WHERE document_id=d.id) AS latest_version, "
           "(SELECT status FROM document_versions WHERE document_id=d.id ORDER BY version_no DESC LIMIT 1) AS latest_status, "
           "(SELECT COUNT(*) FROM signatories s WHERE s.version_id=(SELECT id FROM document_versions WHERE document_id=d.id ORDER BY version_no DESC LIMIT 1)) AS sig_total, "
           "(SELECT COUNT(*) FROM signatories s WHERE s.version_id=(SELECT id FROM document_versions WHERE document_id=d.id ORDER BY version_no DESC LIMIT 1) AND s.signed_at IS NOT NULL) AS sig_done, "
           "EXISTS (SELECT 1 FROM signatories s WHERE s.version_id=(SELECT id FROM document_versions WHERE document_id=d.id ORDER BY version_no DESC LIMIT 1) "
           "AND s.signer_id=? AND s.signed_at IS NULL) AS my_pending_signature "
           "FROM documents d JOIN categories c ON c.id=d.category_id "
           "LEFT JOIN courses co ON co.id=d.course_id WHERE 1=1")
    args: list = []
    if q.get("q"):  # FR-05.1: keyword across title and description
        like = f"%{q['q']}%"
        sql += " AND (d.title LIKE ? OR d.description LIKE ?)"; args += [like, like]
    for param, col in (("category_id", "d.category_id"), ("course_id", "d.course_id"),
                       ("school_year", "d.school_year"), ("semester", "d.semester"),
                       ("document_type", "d.document_type")):
        if q.get(param):
            sql += f" AND {col}=?"; args.append(q[param])
    if q.get("approval_status"):  # FR-05.2
        sql += (" AND EXISTS (SELECT 1 FROM document_versions v WHERE v.document_id=d.id "
                "AND v.version_no=(SELECT MAX(version_no) FROM document_versions WHERE document_id=d.id) AND v.status=?)")
        args.append(q["approval_status"])
    vis, vargs = _visible_clause()
    rows = get_db().execute(sql + vis + " ORDER BY d.school_year DESC, d.title", [g.user["id"]] + args + vargs).fetchall()
    results = []
    for row in rows:
        item = dict(row)
        item["can_update"] = g.user["role"] == "dept_chair" or item["owner_id"] == g.user["id"]
        del item["owner_id"]
        results.append(item)
    return jsonify(results)


def _store(file, doc_id, version_no):
    ext = file.filename.rsplit(".", 1)[-1].lower() if "." in file.filename else ""
    if ext not in current_app.config["ALLOWED_EXTENSIONS"]:
        abort(400, "File type not allowed.")
    data = file.read()
    if not data:
        abort(400, "File is empty.")
    name = f"{doc_id}_v{version_no}_{uuid.uuid4().hex}.{ext}"
    path = Path(current_app.config["UPLOAD_DIR"]) / name
    path.write_bytes(data)
    return str(path), secure_filename(file.filename), hashlib.sha256(data).hexdigest(), len(data)


def _add_version(db, doc_id, file, action):
    n = db.execute(
        "SELECT COALESCE(MAX(version_no),0)+1 FROM document_versions WHERE document_id=?",
        (doc_id,),
    ).fetchone()[0]
    path, orig, sha, size = _store(file, doc_id, n)
    vid = db.execute(
        "INSERT INTO document_versions (document_id, version_no, stored_path, original_name, sha256, size_bytes, uploaded_by) "
        "VALUES (?,?,?,?,?,?,?)",
        (doc_id, n, path, orig, sha, size, g.user["id"]),
    ).lastrowid
    db.execute("UPDATE revision_requests SET resolved=1 WHERE document_id=?", (doc_id,))
    audit(action, "document", doc_id, {"version": n})
    db.commit()
    prune_version_files(db, doc_id, current_app.config["UPLOAD_DIR"])
    convert.ensure_preview(vid, path, orig)
    return vid, n

    raw_doc_id = (request.get_json(silent=True) or request.form).get("document_id")
    try:
        new_doc_id = int(raw_doc_id)
    except (TypeError, ValueError):
        abort(400, "A valid document_id is required.")
    if not new_doc_id or new_doc_id == doc_id:
        abort(400, "A different document_id is required.")
    db = get_db()
    if not db.execute("SELECT 1 FROM requirements WHERE id=?", (rid,)).fetchone():
        abort(404)
    linked = db.execute(
        "SELECT d.owner_id FROM requirement_evidence e JOIN documents d ON d.id=e.document_id "
        "WHERE e.requirement_id=? AND e.document_id=?", (rid, doc_id)).fetchone()
    if not linked:
        abort(404)
    replacement = db.execute("SELECT owner_id FROM documents WHERE id=?", (new_doc_id,)).fetchone()
    if not replacement:
        abort(404)
    if g.user["role"] == "faculty" and (
            linked["owner_id"] != g.user["id"] or replacement["owner_id"] != g.user["id"]):
        abort(403)
    if db.execute("SELECT 1 FROM requirement_evidence WHERE requirement_id=? AND document_id=?",
                  (rid, new_doc_id)).fetchone():
        abort(409, "That document is already linked to this requirement.")
    db.execute("UPDATE requirement_evidence SET document_id=?, linked_by=? "
               "WHERE requirement_id=? AND document_id=?", (new_doc_id, g.user["id"], rid, doc_id))
    _refresh_requirement_status(db, rid)
    audit("requirement.evidence.replace", "requirement", rid,
          {"previous_document": doc_id, "document": new_doc_id})
    db.commit()
    return jsonify(ok=True)

  const data = r.headers.get("content-type")?.includes("json") ? await r.json() : r;
  if (!r.ok) throw new Error(data.error || "Something went wrong.");
  return data;
}

def blank_pdf(rotate=0):
    w = PdfWriter(); p = w.add_blank_page(612, 792)
    if rotate: p.rotate(rotate)
    b = io.BytesIO(); w.write(b); return b.getvalue()

        return c

    def post(self, c, url, **kw):
        kw.setdefault("headers", {})["X-CSRF-Token"] = c.csrf
